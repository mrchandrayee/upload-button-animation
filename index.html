app.css

/* The dome is free: one box with a
   100% width and no top corners.
   Both walls are one arc that touches
   both walls. */
.keel__liquid {
  height: 125%;
  border-radius: 50% 50% 0 0 /
               18.4% 18.4% 0 0;
  transform: translateY(
    calc(var(--p) * .385 -
         var(--p) * 60px)
  );
}
/* The float rides the apex — the same
   term; nothing keeps them in step. */
.keel__float {
  transform:
    translateY(calc((var(--p) * .385
      - var(--p)) * 60px))
    scaleY(calc(1 - .4 * var(--p)));
}

// no requestAnimationFrame anywhere:
// — and —p are REGISTERED numbers
// with transitions, so one write is
// a whole beat.
setP(v) {
  this.root.style
    .setProperty('--p', v.toFixed(4));
}
// The pacing floor — the entire
// honesty policy in one line.
for await (const { loaded, total }
    of readBlob(blob)) {
  const ceiling = clamp01(
    (now() - t0) / runMin
  );
  const shown = Math.min(
    loaded / total, ceiling, 1
  );
  this.p = shown;
}
